# behaves exactly as before, reading Snip/Snip.txt.

# Lines starting with # are comments, and so is " # ..." after a value
# (except on --name=value lines, where # is part of colours). Names are not
# case-sensitive. Every setting the widget has is in this file; a line
# starting with # shows its default, remove the # to change it.

# ── Look ─────────────────────────────────────────────────────────────────
# One of: zune, spotify, apple-music, ipod, xbox, basic, minimal-rect,
#         minimal-square, candy, pixel, space
# OBS shows the new look when the browser source next loads (right-click
# it > Refresh). For one scene with a different look, add ?theme=candy to
# the page address of that browser source (URL mode, file:///...).
theme=zune

# ── Where track info comes from ──────────────────────────────────────────
#   auto    = use the Spotify API if the credentials below work,
#             otherwise fall back to Snip files
#   spotify = Spotify API only
#   snip    = Snip files only
source=auto

# From your app at https://developer.spotify.com/dashboard
spotify_client_id=

# Generated by the setup (setup-spotify.bat / .sh, or tools/spotify-setup.html
# opened in your normal browser).
# Spotify rotates this token on every use; the widget keeps the newest one
# in OBS's browser storage, so only replace this line when you re-authorize.
spotify_refresh_token=

# How quickly a change shows right after something happened (a skip, pause
# or seek), in milliseconds. 2000 is the minimum. Track ends are predicted
# and checked on time whatever this says.
poll_interval=2000

# The longest gap between checks while a track plays undisturbed, in
# milliseconds (never below poll_interval). Raise it to use fewer Spotify
# requests; a mid-track skip or pause then takes up to this long to show.
# See "Rate limits and the request quota" in the README.
poll_interval_max=8000

# ── Updates ──────────────────────────────────────────────────────────────
# When the widget loads it checks for a newer release and runs that instead,
# keeping this file and your colours. off = always run the files as they are.
auto_update=on
# Where releases are read from. Only for a fork; https, or http on this
# computer (127.0.0.1 / localhost).
#update_url=https://masstarvt.github.io/Spotify-Widget/

# ── Colours, sizes, position ─────────────────────────────────────────────

# No " # comment" after these values: a colour like #1DB954 needs its #.

#--title-size=15px
